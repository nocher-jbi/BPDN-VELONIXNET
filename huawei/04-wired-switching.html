Wired Switching

03.1 VLAN

# Batch create
[HUAWEI] vlan batch 10 20 30 99 100 200 1000-1010

# Named VLAN
[HUAWEI] vlan 10
[HUAWEI-vlan10] name MGMT
[HUAWEI-vlan10] description "Management VLAN"
[HUAWEI-vlan10] quit

# VLANIF (L3 Interface)
[HUAWEI] interface vlanif 10
[HUAWEI-Vlanif10] ip address 10.10.10.1 255.255.255.0
[HUAWEI-Vlanif10] vrrp vrid 10 virtual-ip 10.10.10.254
[HUAWEI-Vlanif10] quit

03.2 Port Configuration (Access/Trunk/Hybrid)

# Access port (massal via port-group)
[HUAWEI] port-group access_ports
[HUAWEI-port-group-access_ports] group-member gigabitethernet 0/0/1 to gigabitethernet 0/0/24
[HUAWEI-port-group-access_ports] port link-type access
[HUAWEI-port-group-access_ports] port default vlan 20
[HUAWEI-port-group-access_ports] quit

# Trunk port
[HUAWEI] interface gigabitethernet 0/0/25
[HUAWEI-GigabitEthernet0/0/25] port link-type trunk
[HUAWEI-GigabitEthernet0/0/25] port trunk allow-pass vlan 10 20 30 99 100 200
[HUAWEI-GigabitEthernet0/0/25] port trunk pvid vlan 99
[HUAWEI-GigabitEthernet0/0/25] quit

# Hybrid (untuk QinQ/voice VLAN)
[HUAWEI] interface gigabitethernet 0/0/1
[HUAWEI-GigabitEthernet0/0/1] port link-type hybrid
[HUAWEI-GigabitEthernet0/0/1] port hybrid tagged vlan 200
[HUAWEI-GigabitEthernet0/0/1] port hybrid untagged vlan 20
[HUAWEI-GigabitEthernet0/0/1] quit

03.5 STP / RSTP / MSTP / VBST / ERPS

# MSTP (Recommended)
[HUAWEI] stp mode mstp
[HUAWEI] stp region-configuration
[HUAWEI-mst-region] region-name COMPANY_NET
[HUAWEI-mst-region] revision-level 1
[HUAWEI-mst-region] instance 1 vlan 10 20 30
[HUAWEI-mst-region] instance 2 vlan 100 200
[HUAWEI-mst-region] active region-configuration
[HUAWEI-mst-region] quit

[HUAWEI] stp instance 1 root primary
[HUAWEI] stp instance 2 root secondary

# Edge Port (PortFast)
[HUAWEI] port-group access_ports
[HUAWEI-port-group-access_ports] stp edged-port enable
[HUAWEI-port-group-access_ports] stp bpdu-filter enable
[HUAWEI-port-group-access_ports] quit

# ERPS (G.8032 Ring Protection)
[HUAWEI] erps ring 1
[HUAWEI-erps-ring1] control-vlan 100
[HUAWEI-erps-ring1] protected-instance 1 to 10
[HUAWEI-erps-ring1] wtr-timer 5
[HUAWEI-erps-ring1] quit
[HUAWEI] interface gigabitethernet 0/0/25
[HUAWEI-GigabitEthernet0/0/25] erps ring 1 rpl owner
[HUAWEI-GigabitEthernet0/0/25] quit
[HUAWEI] interface gigabitethernet 0/0/26
[HUAWEI-GigabitEthernet0/0/26] erps ring 1 rpl neighbor
[HUAWEI-GigabitEthernet0/0/26] quit
[HUAWEI] erps ring 1 enable
