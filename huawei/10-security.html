09.2 ACL (IPv4/IPv6/L2)

# Basic ACL (Src IP)
[HUAWEI] acl 2000
[HUAWEI-acl-basic-2000] rule 5 permit source 10.10.10.0 0.0.0.255
[HUAWEI-acl-basic-2000] rule 10 deny
[HUAWEI-acl-basic-2000] quit

# Advanced ACL (L3/L4)
[HUAWEI] acl 3000
[HUAWEI-acl-adv-3000] rule 5 permit tcp source 192.168.20.0 0.0.0.255 destination-port eq 443
[HUAWEI-acl-adv-3000] rule 10 permit udp source 192.168.20.0 0.0.0.255 destination-port eq 53
[HUAWEI-acl-adv-3000] rule 15 deny ip
[HUAWEI-acl-adv-3000] quit

# Apply to interface
[HUAWEI] interface vlanif 20
[HUAWEI-Vlanif20] traffic-filter inbound acl 3000
[HUAWEI-Vlanif20] quit

# Apply to VTY
[HUAWEI] user-interface vty 0 4
[HUAWEI-ui-vty0-4] acl 2000 inbound
[HUAWEI-ui-vty0-4] quit
