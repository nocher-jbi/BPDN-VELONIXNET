11.1 NETCONF/YANG (Model-Driven Automation)

# Enable NETCONF over SSH (port 830)
[HUAWEI] netconf
[HUAWEI-netconf] ssh server enable
[HUAWEI-netconf] quit

# YANG Models: ietf-interfaces, ietf-vlan, huawei-qos, huawei-acl, huawei-bgp, huawei-mpls
# Get capabilities:
# <hello><capabilities><capability>urn:ietf:params:netconf:capability:yang-library:1.0</capability></capabilities></hello>

Common NETCONF RPC Examples

<!-- Get running config -->
<get-config>
  <source><running/></source>
</get-config>

<!-- Edit config: Add VLAN -->
<edit-config>
  <target><running/></target>
  <config>
    <vlans xmlns="urn:ietf:params:xml:ns:yang:ietf-vlan">
      <vlan>
        <vlan-id>100</vlan-id>
        <name>SERVER</name>
        <status>active</status>
      </vlan>
    </vlans>
  </config>
</edit-config>

<!-- Get interface statistics -->
<get>
  <filter type="subtree">
    <interfaces-state xmlns="urn:ietf:params:xml:ns:yang:ietf-interfaces">
      <interface>
        <name>GigabitEthernet0/0/1</name>
      </interface>
    </interfaces-state>
  </filter>
</get>
