Routing

05.1 Static Routes

[HUAWEI] ip route-static 0.0.0.0 0.0.0.0 10.10.10.1          # Default
[HUAWEI] ip route-static 192.168.100.0 255.255.255.0 10.10.10.50   # Specific
[HUAWEI] ip route-static 10.0.0.0 8 10.10.10.100 preference 80   # Floating static
[HUAWEI] ip route-static vpn-instance MGMT 0.0.0.0 0.0.0.0 10.10.10.1  # VRF

05.2 OSPF (Single / Multi-Area)

[HUAWEI] ospf 1 router-id 10.10.10.2
[HUAWEI-ospf-1] silent-interface vlanif 10
[HUAWEI-ospf-1] area 0.0.0.0
[HUAWEI-ospf-1-area-0.0.0.0] network 10.10.10.0 0.0.0.255
[HUAWEI-ospf-1-area-0.0.0.0] network 192.168.20.0 0.0.0.255
[HUAWEI-ospf-1-area-0.0.0.0] quit

[HUAWEI-ospf-1] area 0.0.0.1
[HUAWEI-ospf-1-area-0.0.0.1] network 192.168.30.0 0.0.0.255
[HUAWEI-ospf-1-area-0.0.0.1] quit

[HUAWEI-ospf-1] default-route-advertise always
[HUAWEI-ospf-1] quit

05.3 IS-IS (Large Scale / SP)

[HUAWEI] isis 1
[HUAWEI-isis-1] is-level level-1-2
[HUAWEI-isis-1] network-entity 10.0000.0000.0001.00
[HUAWEI-isis-1] quit

[HUAWEI] interface vlanif 10
[HUAWEI-Vlanif10] isis enable 1
[HUAWEI-Vlanif10] quit

05.4 BGP (Internet / DC / L3VPN)

[HUAWEI] bgp 65001
[HUAWEI-bgp] router-id 10.10.10.2
[HUAWEI-bgp] peer 10.10.10.254 as-number 65000
[HUAWEI-bgp] peer 10.10.10.254 connect-interface loopback0

# IPv4 Unicast
[HUAWEI-bgp] ipv4-family unicast
[HUAWEI-bgp-af-ipv4] peer 10.10.10.254 enable
[HUAWEI-bgp-af-ipv4] network 192.168.20.0 255.255.255.0
[HUAWEI-bgp-af-ipv4] quit

# EVPN (VXLAN)
[HUAWEI-bgp] l2vpn-family evpn
[HUAWEI-bgp-af-evpn] peer 10.10.10.254 enable
[HUAWEI-bgp-af-evpn] peer 10.10.10.254 advertise encap-type vxlan
[HUAWEI-bgp-af-evpn] quit

[HUAWEI-bgp] quit

05.5 Route Policy / Prefix List

[HUAWEI] ip prefix-list DEFAULT_ROUTE permit 0.0.0.0 0
[HUAWEI] route-policy EXPORT_OSPF permit node 10
[HUAWEI-route-policy] if-match ip-prefix DEFAULT_ROUTE
[HUAWEI-route-policy] apply cost 100
[HUAWEI-route-policy] quit

05.6 VRF / VPN Instance

[HUAWEI] ip vpn-instance MGMT
[HUAWEI-vpn-instance-MGMT] ipv4-family
[HUAWEI-vpn-instance-MGMT-ipv4] route-distinguisher 100:10
[HUAWEI-vpn-instance-MGMT-ipv4] vpn-target 100:10 export-extcommunity
[HUAWEI-vpn-instance-MGMT-ipv4] vpn-target 100:10 import-extcommunity
[HUAWEI-vpn-instance-MGMT-ipv4] quit

[HUAWEI] interface Meth0/0/0
[HUAWEI-MEth0/0/0] ip binding vpn-instance MGMT
[HUAWEI-MEth0/0/0] ip address 10.10.10.2 255.255.255.0
